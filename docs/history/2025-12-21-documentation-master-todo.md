**Алгоритм обнаружения (В ЦЕЛЕВОМ ПРОЕКТЕ):**
```python
def find_artifact(artifact_type: str) -> Path | None:
    # 1. Проверить .pipeline-state.json (в целевом проекте)
    state = read_json(".pipeline-state.json")
    if state and state.artifacts.get(artifact_type):
        return Path(state.artifacts[artifact_type])

    # 2. Glob по паттернам В ЦЕЛЕВОМ ПРОЕКТЕ
    patterns = {
        "prd": "ai-docs/docs/prd/*-prd.md",
        "plan": "ai-docs/docs/architecture/*-plan.md",
        "feature_plan": "ai-docs/docs/plans/*-plan.md"
    }
    files = glob(patterns[artifact_type])
    if files:
        return max(files, key=lambda f: f.stat().st_mtime)

    return None
```
